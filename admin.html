async function startLive() { const titleInput = document.getElementById("liveTitle"); const descriptionInput = document.getElementById("liveDescription"); const title = titleInput?.value.trim(); const description = descriptionInput?.value.trim() || ""; if (!title) { alert("Enter a live title first."); return; } // تأكد أن المستخدم مسجل دخول const { data: { user }, error: authError } = await supabaseClient.auth.getUser(); if (authError || !user) { alert("You must be logged in as an administrator."); return; } // تأكد أن المستخدم Admin const { data: adminRole, error: adminError } = await supabaseClient .from("admin_roles") .select("role") .eq("user_id", user.id) .maybeSingle(); if (adminError) { console.error(adminError); alert("Could not verify administrator permissions."); return; } if (!adminRole || adminRole.role !== "admin") { alert("This account is not an administrator."); return; } // إيقاف أي بث قديم const { error: stopError } = await supabaseClient .from("live_sessions") .update({ is_live: false, ended_at: new Date().toISOString() }) .eq("is_live", true); if (stopError) { console.error(stopError); alert("Could not close previous live session."); return; } // إنشاء البث الجديد const { data, error } = await supabaseClient .from("live_sessions") .insert({ title: title, description: description, stream_url: "", thumbnail_url: "", is_live: true, started_at: new Date().toISOString() }) .select() .single(); if (error) { console.error("Start Live Error:", error); alert("Could not start live: " + error.message); return; } // تحديث واجهة الأدمن const dot = document.getElementById("liveDot"); const status = document.getElementById("liveStatus"); if (dot) { dot.classList.add("live"); } if (status) { status.textContent = "LIVE — " + title; } alert("🔴 Live started successfully!"); }
